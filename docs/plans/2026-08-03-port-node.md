## Global Constraints

- Node ≥ 20, TypeScript `strict: true`, sin dependencias nativas (nada de node-gyp).
- Dominio en español: `Factura`, `claveAcceso`, `Ambiente`, igual que el XML del SRI.
- Montos = `string` con 2 decimales (`"100.00"`); aritmética solo en centavos enteros (utils/money).
- Rechazo del SRI NO lanza excepción: se refleja en `EmissionResult.status`.
- Fixtures golden generados con el paquete PHP (PHP 8.4 disponible en la máquina).
- Commits frecuentes con mensajes convencionales (`feat:`, `test:`, `chore:`).
- Cobertura: cada módulo portado lleva sus tests vitest en el mismo task.

## Contratos compartidos (canónicos — no desviarse)

```ts
// catalogs/index.ts
export enum Ambiente { Pruebas = '1', Produccion = '2' }
export enum TipoComprobante {
  Factura = '01', LiquidacionCompra = '03', NotaCredito = '04',
  NotaDebito = '05', GuiaRemision = '06', Retencion = '07',
}
export enum TipoEmision { Normal = '1' }

// emission/index.ts
export type EmissionStatus = 'AUTORIZADO' | 'RECHAZADO' | 'EN_PROCESO' | 'ERROR';
export type RejectionStage = 'RECEPCION' | 'AUTORIZACION';
export interface Message {
  identificador: string; mensaje: string;
  tipo?: string; informacionAdicional?: string;
}
export interface EmissionResult {
  status: EmissionStatus; claveAcceso: string; signedXml: string;
  numeroAutorizacion?: string; fechaAutorizacion?: string; authorizedXml?: string;
  messages: Message[]; rejectedStage?: RejectionStage;
}

// transport/types.ts
export interface ReceptionOutcome { estado: string; mensajes: Message[] }
export interface AuthorizationOutcome {
  estado: string; numeroAutorizacion?: string; fechaAutorizacion?: string;
  comprobante?: string; mensajes: Message[];
}
export interface SriTransport {
  enviar(signedXml: string, ambiente: Ambiente, opts?: { signal?: AbortSignal }): Promise<ReceptionOutcome>;
  autorizar(claveAcceso: string, ambiente: Ambiente, opts?: { signal?: AbortSignal }): Promise<AuthorizationOutcome>;
}

// signing
export interface Certificate { certPem: string; privateKeyPem: string; extraCerts: string[] }
export function loadCertificate(p12: Uint8Array, password: string): Certificate;
export interface SignatureOptions { digestAlgorithm?: 'sha1'; description?: string }
export interface Clock { now(): Date }
export class XadesSigner {
  constructor(options?: SignatureOptions, clock?: Clock);
  sign(xml: string, cert: Certificate): string;
}

// documents — discriminante `tipo`; montos string "0.00"
// (estructuras completas en Task 4)
export type Comprobante = Factura | LiquidacionCompra | NotaCredito
  | NotaDebito | GuiaRemision | Retencion;

// xml — un serializador por tipo
export interface XmlSerializer<D> { serialize(doc: D, claveAcceso: string): string }

// sri-client.ts
export interface SriClientOptions {
  ambiente: Ambiente; certificate: Certificate;
  transport?: SriTransport;          // default FetchSoapTransport
  signer?: XadesSigner; clock?: Clock;
  validate?: boolean;                // default true (zod + business)
}
export class SriClient {
  constructor(options: SriClientOptions);
  emit(doc: Comprobante, claveAcceso?: string): Promise<EmissionResult>;
  authorize(claveAcceso: string): Promise<AuthorizationOutcome>;
  sign(xml: string): string;
}

// errors/index.ts
export class SriError extends Error { readonly code: string }
export class ValidationError extends SriError { code: 'VALIDATION'; errors: string[] }
export class CertificateError extends SriError { code: 'CERTIFICATE' }
export class SignatureError extends SriError { code: 'SIGNATURE' }
export class CommunicationError extends SriError { code: 'COMMUNICATION'; cause?: unknown }

// batch
export type ComprobanteState = 'PENDING' | 'SENT' | 'AUTHORIZED' | 'REJECTED' | 'IN_PROCESS' | 'FAILED';
export interface BatchItem {
  claveAcceso: string; signedXml: string; state: ComprobanteState;
  attempts: number; numeroAutorizacion?: string; authorizedXml?: string; messages: Message[];
}
export class BatchEmitter {
  constructor(opts: { ambiente: Ambiente; transport?: SriTransport; retryPolicy?: RetryPolicy });
  add(claveAcceso: string, signedXml: string): void;   // idempotente
  run(opts?: { maxPasses?: number }): Promise<void>;   // reanudable
  status(): Record<ComprobanteState, number>;
  result(claveAcceso: string): BatchItem | undefined;
}
```

Convención de archivos test: `packages/sri-ec/test/<modulo>.test.ts`. Ejecutar con
`npm test -w @amephia/sri-ec` (vitest run).
